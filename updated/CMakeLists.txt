cmake_minimum_required(VERSION 3.16)
project(marker_tracking_updated LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d videoio highgui)
add_executable(marker_tracker src/main.cpp src/SearchTree.cpp)
target_include_directories(marker_tracker PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(marker_tracker PRIVATE ${OpenCV_LIBS})
target_compile_options(marker_tracker PRIVATE -Wall -Wextra -Wno-sign-compare -Wno-unused-variable -Wno-unused-parameter)
option(MARKER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MARKER_SANITIZE)
 target_compile_options(marker_tracker PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(marker_tracker PRIVATE -fsanitize=address,undefined)
endif()
include(CTest)
add_test(NAME tree_regression COMMAND marker_tracker selftest)
